\begin{frame}{Notation: Spaces and Measures}
\textbf{Measure Spaces:}
\begin{itemize}
    \item $\Pp(X)$: \textbf{probability measures} on $X$ \quad ($\mu(X) = 1$, $\mu \geq 0$)
    \item $\Mm(X)$: \textbf{finite signed measures} on $X$ \quad ($\mu = \mu^+ - \mu^-$)
    \item $\Mm_+(X)$: \textbf{positive finite measures} on $X$ \quad ($\mu \geq 0$, $\mu(X) < \infty$)
\end{itemize}

\[
\underbrace{\delta_a(A)}_{\text{Dirac at } a} := \begin{cases} 1 & \text{if } a \in A \\ 0 & \text{if } a \notin A \end{cases}

\begin{frame}{Notation: Indicator and Lebesgue Measure}
\textbf{Indicator Function:}
\[
\mathbb{1}_A(x) = \begin{cases} 1 & \text{if } x \in A \\ 0 & \text{if } x \notin A \end{cases}
\]

\vspace{0.3cm}
\textbf{Lebesgue Measure:}
\[
|A| = \Ld(A) = \int_A dx \quad \text{(volume of set } A \subset \R^d\text{)}
\]

\vspace{0.3cm}
\textbf{Density:} $f \cdot \mu$ denotes measure with density $f$ w.r.t. $\mu$:
\[
\underbrace{(f \cdot \mu)(A)}_{\text{measure of } A} = \int_A \underbrace{f(x)}_{\text{density}} \, \underbrace{d\mu(x)}_{\text{base measure}}
\]

\vspace{0.3cm}
\textbf{Restriction:} $\mu \llcorner A := \mathbb{1}_A \cdot \mu$ (restrict $\mu$ to set $A$)
\end{frame}

\begin{frame}{Notation: Costs and Distances}
\textbf{Monge Cost:} For transport map $T: X \to Y$
\[
M(T) := \int_X \underbrace{c(x, T(x))}_{\text{cost per unit}} \, \underbrace{d\mu(x)}_{\text{mass at } x}
\]

\vspace{0.3cm}
\textbf{Kantorovich Cost:} For transport plan $\gamma \in \Pi(\mu,\nu)$
\[
K(\gamma) := \int_{X \times Y} \underbrace{c(x,y)}_{\text{cost } x \to y} \, \underbrace{d\gamma(x,y)}_{\text{mass } x \to y}
\]

\vspace{0.3cm}
\textbf{Wasserstein Distance:}
\[
W_p(\mu,\nu) := \left( \inf_{\gamma \in \Pi(\mu,\nu)} \int |x-y|^p \, d\gamma \right)^{1/p}
\]

\vspace{0.2cm}
\textbf{Minimal Transport Cost:}
\[
\mathcal{T}_c(\mu,\nu) := \inf_{\gamma \in \Pi(\mu,\nu)} K(\gamma)
\]
\end{frame}

\section{The Monge Problem}
%==============================================================================

\begin{frame}{The Monge Problem (1781)}
Given $\mu \in \Pp(X)$, $\nu \in \Pp(Y)$, and cost $c: X \times Y \to [0, +\infty]$:

\vspace{0.3cm}
\[
\boxed{\textbf{(MP)} \quad \inf \left\{ M(T) := \int_X \underbrace{c(x, T(x))}_{\substack{\text{cost to move}\\ \text{unit mass } x \to T(x)}} \, d\mu(x) \;:\; \underbrace{T_\# \mu = \nu}_{\substack{\text{mass balance:}\\ \text{all mass arrives correctly}}} \right\}}
\]

\begin{frame}{Push-forward Measure}
\textbf{Definition:} For map $T: X \to Y$ and measure $\mu$ on $X$:
\[
\underbrace{(T_\# \mu)(A)}_{\text{measure of } A \text{ under } T_\#\mu} := \mu\bigl(\underbrace{T^{-1}(A)}_{\text{preimage of } A}\bigr)
\]

\vspace{0.3cm}
\textbf{Integral characterization:} For any measurable $\phi: Y \to \R$:
\[
\int_Y \underbrace{\phi(y)}_{\text{test function}} \, d(T_\# \mu)(y) = \int_X \underbrace{\phi(T(x))}_{\text{composition}} \, d\mu(x)
\]

\begin{frame}{Jacobian Equation}
When $\mu = f \cdot \Ld$ and $\nu = g \cdot \Ld$ have densities, and $T$ is a diffeomorphism:

\vspace{0.3cm}
\[
\boxed{\underbrace{g(T(x))}_{\text{target density at } T(x)} \cdot \underbrace{|\det DT(x)|}_{\text{volume change factor}} = \underbrace{f(x)}_{\text{source density at } x}}
\]

\vspace{0.3cm}

\begin{frame}{Difficulty: Mass Cannot Split}
\textbf{Problem:} A transport \emph{map} $T$ sends each point to exactly one destination.

\vspace{0.3cm}
\textbf{Example:} $\mu = \delta_0$, $\nu = \frac{1}{2}\delta_{-1} + \frac{1}{2}\delta_{1}$

\vspace{0.3cm}

$\Rightarrow$ \textbf{No transport map exists!}

\vspace{0.2cm}
\textbf{Other issues with Monge problem:}
\begin{itemize}
    \item Constraint $T_\# \mu = \nu$ is \textbf{nonlinear} in $T$
    \item Set of admissible maps is \textbf{not closed} under weak limits
\end{itemize}
\end{frame}

\begin{frame}{Exercise 1: Push-forward Computation}
\textbf{Problem:} Let $\mu$ be the uniform distribution on $[0,1]$ (i.e., $\mu = \mathbb{1}_{[0,1]} \cdot \Ld$).

Let $T(x) = 2x$.

\vspace{0.5cm}
\textbf{Questions:}
\begin{enumerate}
    \item[(a)] What is $T_\# \mu$? Find its density.
    \item[(b)] Verify your answer using the integral characterization:
    \[
    \int_\R \phi(y) \, d(T_\# \mu)(y) = \int_0^1 \phi(2x) \, dx
    \]
    \item[(c)] What is the Jacobian equation in this case?
\end{enumerate}

\vspace{0.3cm}
\textit{Hint:} For (a), determine the support of $T_\# \mu$ and use mass conservation.
\end{frame}

%==============================================================================
\begin{frame}{Solution 1}
\begin{columns}
\begin{column}{0.55\textwidth}
\textbf{(a)} $T(x) = 2x$ maps $[0,1]$ to $[0,2]$.

By mass conservation: total mass $= 1$.

If $T_\# \mu = g \cdot \Ld$, then $\int_0^2 g(y) \, dy = 1$.

Using Jacobian: $g(2x) \cdot 2 = 1$, so $g(y) = \frac{1}{2}$.

\[
\boxed{T_\# \mu = \frac{1}{2} \mathbb{1}_{[0,2]} \cdot \Ld}
\]

\vspace{0.3cm}
\textbf{(b)} Verification:
\[
\int_0^2 \phi(y) \cdot \frac{1}{2} \, dy \stackrel{y=2x}{=} \int_0^1 \phi(2x) \, dx \;\checkmark
\]

\vspace{0.2cm}
\textbf{(c)} Jacobian: $g(T(x)) |T'(x)| = f(x)$
\[
\frac{1}{2} \cdot 2 = 1 \;\checkmark
\]
\end{column}

\section{Kantorovich Relaxation}
%==============================================================================

\begin{frame}{Kantorovich's Idea (1942)}
\textbf{Key insight:} Replace transport \emph{maps} with transport \emph{plans}.

\vspace{0.3cm}
A \textbf{transport plan} $\gamma \in \Pp(X \times Y)$ specifies:
\[
\gamma(A \times B) = \text{``mass moving from region } A \text{ to region } B\text{''}
\]

\vspace{0.3cm}
\begin{center}
\begin{tikzpicture}[scale=0.8]
    % Monge
    \begin{scope}[xshift=-3cm]
        \fill[blue!70] (0,0) circle (4pt);
        \draw[-{Stealth},thick] (0.15,0) -- (1.85,0);
        \fill[red!70] (2,0) circle (4pt);
        \node at (1,-0.8) {\small \textbf{Monge:} 1-to-1};
    \end{scope}

    % Kantorovich (splitting)
    \begin{scope}[xshift=3cm]
        \fill[blue!70] (0,0) circle (4pt);
        \draw[-{Stealth},thick] (0.1,0.1) -- (1.85,0.5);
        \draw[-{Stealth},thick] (0.1,-0.1) -- (1.85,-0.5);
        \fill[red!70] (2,0.5) circle (3pt);
        \fill[red!70] (2,-0.5) circle (3pt);
        \node at (1,-1) {\small \textbf{Kantorovich:} splitting OK};
    \end{scope}
\end{tikzpicture}
\end{center}

\vspace{0.3cm}
\textbf{Marginal constraints:}
\[
\underbrace{(\pi_x)_\# \gamma = \mu}_{\text{total mass leaving each source}} \quad \text{and} \quad \underbrace{(\pi_y)_\# \gamma = \nu}_{\text{total mass arriving at each target}}
\]
\end{frame}

%==============================================================================
\begin{frame}{Transport Plan Definition}
\textbf{Set of transport plans:}
\[
\boxed{\Pi(\mu, \nu) := \left\{ \gamma \in \Pp(X \times Y) : \underbrace{(\pi_x)_\# \gamma = \mu}_{\text{1st marginal}}, \; \underbrace{(\pi_y)_\# \gamma = \nu}_{\text{2nd marginal}} \right\}}
\]

\vspace{0.3cm}
where $\pi_x(x,y) = x$ and $\pi_y(x,y) = y$ are projections.

\vspace{0.3cm}
\textbf{In terms of integrals:}
\[
\int_{X \times Y} \phi(x) \, d\gamma = \int_X \phi \, d\mu \quad \text{and} \quad \int_{X \times Y} \psi(y) \, d\gamma = \int_Y \psi \, d\nu
\]

\vspace{0.3cm}
\textbf{Key property:} $\Pi(\mu,\nu) \neq \emptyset$ always!

\vspace{0.2cm}
\textit{Example:} The product measure $\mu \otimes \nu \in \Pi(\mu,\nu)$ (``independent coupling'').
\end{frame}

%==============================================================================
\begin{frame}{Kantorovich Problem (KP)}
\[
\boxed{\textbf{(KP)} \quad \inf \left\{ K(\gamma) := \int_{X \times Y} \underbrace{c(x,y)}_{\text{cost per unit}} \, \underbrace{d\gamma(x,y)}_{\text{mass from } x \text{ to } y} \;:\; \gamma \in \Pi(\mu, \nu) \right\}}
\]

\vspace{0.4cm}
\textbf{Key advantages over (MP):}

\begin{center}
\begin{tabular}{l|c|c}
& \textbf{Monge (MP)} & \textbf{Kantorovich (KP)} \\
\hline
Objective & Nonlinear in $T$ & \textbf{Linear} in $\gamma$ \\
Constraints & Nonlinear $T_\#\mu = \nu$ & \textbf{Linear} marginals \\
Existence & May not exist & \textbf{Always exists} \\
Mass splitting & Not allowed & \textbf{Allowed} \\
Convexity & Non-convex & \textbf{Convex} (LP)
\end{tabular}
\end{center}

\vspace{0.3cm}
$\Rightarrow$ \textbf{Linear Programming} problem!
\end{frame}

\begin{frame}{Transport Plan Visualization}
\begin{columns}
\begin{column}{0.5\textwidth}

\begin{frame}{Transport Plan from a Map}
If $T: X \to Y$ satisfies $T_\# \mu = \nu$, define:
\[
\gamma_T := (\text{id}, T)_\# \mu
\]

This means: $\gamma_T$ is concentrated on the \textbf{graph} $\{(x, T(x)) : x \in X\}$.

\[
\gamma_T(A \times B) = \mu\bigl(\{x \in A : T(x) \in B\}\bigr)

\vspace{0.3cm}

\[
K(\gamma_T) = \int_{X \times Y} c(x,y) \, d\gamma_T = \int_X c(x, T(x)) \, d\mu(x) = M(T)
\]

\vspace{0.3cm}
$\Rightarrow$ \textbf{Optimal maps give optimal plans} (when they exist)
\[
\min(\text{KP}) \leq \min(\text{MP}) \quad \text{(equality under good conditions)}
\]
\end{frame}

\begin{frame}{Box 1.1: Weierstrass Theorem}
\begin{block}{Weierstrass Theorem (Direct Method)}
Let $X$ be a topological space and $f: X \to \R \cup \{+\infty\}$.

If:
\begin{enumerate}
    \item $X$ is \textbf{compact}
    \item $f$ is \textbf{lower semi-continuous} (l.s.c.)
\end{enumerate}

Then $f$ attains its minimum: $\exists \bar{x} \in X$ with $f(\bar{x}) = \inf_X f$.
\end{block}

\vspace{0.3cm}
\textbf{Proof outline:}
\begin{enumerate}
    \item Take minimizing sequence: $f(x_n) \to \ell := \inf f$
    \item By compactness: $x_{n_k} \to \bar{x}$ for some subsequence
    \item By l.s.c.: $f(\bar{x}) \leq \liminf_{k} f(x_{n_k}) = \ell$
    \item Since $\ell = \inf f$: $f(\bar{x}) = \ell = \min f$ \quad $\square$
\end{enumerate}
\end{frame}

%==============================================================================
\begin{frame}{Lower Semi-Continuity (l.s.c.)}
\begin{columns}
\begin{column}{0.55\textwidth}
\textbf{Definition:} $f: X \to \R \cup \{+\infty\}$ is \textbf{l.s.c.} if:
\[
\boxed{f(x) \leq \liminf_{n \to \infty} f(x_n) \quad \text{whenever } x_n \to x}
\]

\vspace{0.3cm}
\textbf{Equivalently:}
\begin{itemize}
    \item $\{x : f(x) \leq \alpha\}$ is closed for all $\alpha \in \R$
    \item Epigraph $\{(x,t) : f(x) \leq t\}$ is closed
\end{itemize}

\vspace{0.3cm}
\textbf{Intuition:}
\begin{itemize}
    \item ``Jumps down'' are forbidden
    \item ``Jumps up'' are allowed
\end{itemize}
\end{column}

\begin{frame}{Box 1.2: Weak Compactness in Dual Spaces}
\textbf{Setting:} $V$ a Banach space, $V^*$ its dual.

\vspace{0.3cm}
\textbf{Definition (Weak-* convergence):} $\phi_n \stackrel{*}{\rightharpoonup} \phi$ in $V^*$ if
\[
\langle \phi_n, v \rangle \to \langle \phi, v \rangle \quad \forall v \in V
\]

\vspace{0.3cm}
\begin{block}{Banach-Alaoglu Theorem}
The closed unit ball $\{\phi \in V^* : \|\phi\| \leq 1\}$ is \textbf{weak-* compact}.
\end{block}

\vspace{0.3cm}
\textbf{Consequence:} Any bounded sequence in $V^*$ has a weak-* convergent subsequence.

\vspace{0.3cm}
\textbf{Application to OT:} Measures can be viewed as elements of $\Cb(X)^*$.
\[
\mu_n \text{ bounded } \Rightarrow \exists \text{ weakly convergent subsequence}
\]
\end{frame}

\begin{frame}{Box 1.3: Duality Between $C_0$ and $\mathcal{M}$}
\textbf{Riesz Representation Theorem:}

Let $X$ be a locally compact Hausdorff space.

\vspace{0.3cm}
\[
\boxed{\Czero(X)^* \cong \Mm(X)}
\]

\vspace{0.3cm}
Every continuous linear functional on $\Czero(X)$ is given by:
\[
\Lambda(\phi) = \int_X \phi \, d\mu
\]
for a unique $\mu \in \Mm(X)$ (signed finite measure).

\vspace{0.4cm}
\textbf{Narrow convergence:} $\mu_n \rightharpoonup \mu$ if
\[
\int \phi \, d\mu_n \to \int \phi \, d\mu \quad \forall \phi \in \Cb(X)
\]

\vspace{0.2cm}
\textbf{Note:} For probability measures, this is the standard weak convergence.
\end{frame}

\begin{frame}{Box 1.4: Prokhorov Theorem}
\begin{columns}
\begin{column}{0.55\textwidth}
\textbf{Definition (Tightness):}
A family $\{\mu_n\} \subset \Pp(X)$ is \textbf{tight} if:
\[
\forall \varepsilon > 0, \; \exists K \text{ compact}:
\]
\[
\mu_n(X \setminus K) < \varepsilon \quad \forall n
\]

\vspace{0.3cm}
\begin{block}{Prokhorov's Theorem}
On a Polish space (complete separable metric):
\[
\{\mu_n\} \text{ tight} \;\Leftrightarrow\; \exists \text{ weakly conv. subseq.}
\]
\end{block}

\vspace{0.3cm}
\textbf{Key fact:} A single probability measure is always tight.

\textbf{Application:} $\Pi(\mu,\nu)$ is tight if $\mu, \nu$ are tight.
\end{column}

\begin{frame}{Box 1.5: l.s.c. as Suprema of Lipschitz Functions}
\textbf{Theorem:} If $f: X \to \R \cup \{+\infty\}$ is l.s.c. and bounded below, then:
\[
f(x) = \sup \{ g(x) : g \leq f, \; g \text{ Lipschitz} \}
\]

\vspace{0.3cm}
\textbf{Consequence:} If $\mu_n \rightharpoonup \mu$ and $f$ is l.s.c., then:
\[
\int f \, d\mu \leq \liminf_{n \to \infty} \int f \, d\mu_n
\]

\vspace{0.3cm}
\textbf{Application to OT:}
\begin{itemize}
    \item If cost $c$ is l.s.c., then $K(\gamma) = \int c \, d\gamma$ is l.s.c. in $\gamma$
    \item Combined with compactness $\Rightarrow$ existence of optimal plan
\end{itemize}
\end{frame}

%==============================================================================
\begin{frame}{Box 1.6: Lusin's Theorem}
\textbf{Theorem (Lusin):} Let $\mu$ be a finite measure on $X$ and $f: X \to \R$ measurable.

For every $\varepsilon > 0$, there exists $K \subset X$ compact such that:
\begin{enumerate}
    \item $\mu(X \setminus K) < \varepsilon$
    \item $f|_K$ is continuous
\end{enumerate}

\vspace{0.4cm}
\textbf{Interpretation:} Every measurable function is ``almost continuous.''

\vspace{0.4cm}
\textbf{Strong Lusin:} Given $\varepsilon > 0$, there exists $g \in C_c(X)$ with:
\[
\mu(\{x : f(x) \neq g(x)\}) < \varepsilon
\]

\vspace{0.3cm}
\textbf{Use in OT:} Approximating transport maps by continuous maps.
\end{frame}

%==============================================================================
\begin{frame}{Existence Theorem: Compact Case}
\begin{block}{Theorem 1.4}
Let $X, Y$ be \textbf{compact} metric spaces, $\mu \in \Pp(X)$, $\nu \in \Pp(Y)$, and $c: X \times Y \to \R$ \textbf{continuous}. Then (KP) admits a solution.
\end{block}

\vspace{0.3cm}
\textbf{Proof:}
\begin{enumerate}
    \item \textbf{$\Pi(\mu,\nu)$ is non-empty:} $\mu \otimes \nu \in \Pi(\mu,\nu)$ \checkmark

    \item \textbf{$\Pi(\mu,\nu)$ is compact:}
    \begin{itemize}
        \item Bounded (probability measures)
        \item Closed under weak convergence (marginals preserved)
        \item By Prokhorov: weakly compact
    \end{itemize}

    \item \textbf{$K(\gamma)$ is continuous:}
    \begin{itemize}
        \item $c$ continuous, $\gamma_n \rightharpoonup \gamma$ $\Rightarrow$ $\int c \, d\gamma_n \to \int c \, d\gamma$
    \end{itemize}

    \item Apply Weierstrass theorem $\Rightarrow$ minimum exists. \quad $\square$
\end{enumerate}
\end{frame}

%==============================================================================
\begin{frame}{Existence: General Case}
\begin{block}{Lemma 1.6}
If $f: X \to \R \cup \{+\infty\}$ is l.s.c. and bounded below, then
$J(\mu) := \int f \, d\mu$ is l.s.c. for weak convergence.
\end{block}

\vspace{0.3cm}
\begin{block}{Theorem 1.5}
$X, Y$ compact, $c$ \textbf{l.s.c.} $\Rightarrow$ (KP) has a solution.
\end{block}

\vspace{0.3cm}
\begin{block}{Theorem 1.7}
$X, Y$ \textbf{Polish}, $c: X \times Y \to [0, +\infty]$ l.s.c. $\Rightarrow$ (KP) has a solution.
\end{block}

\vspace{0.3cm}
\textbf{Proof idea for Theorem 1.7:}
\begin{itemize}
    \item $\gamma_n \in \Pi(\mu,\nu)$ tight because $\mu, \nu$ tight
    \item Prokhorov $\Rightarrow$ weak compactness
    \item Lemma 1.6 $\Rightarrow$ l.s.c. of $K(\gamma)$
    \item Weierstrass $\Rightarrow$ minimum exists
\end{itemize}
\end{frame}

\section{Duality}
%==============================================================================

\begin{frame}{Why Duality?}
(KP) is a \textbf{linear optimization} problem:
\begin{itemize}
    \item Objective: $\min \int c \, d\gamma$ is linear in $\gamma$
    \item Constraints: marginals are linear in $\gamma$
\end{itemize}

\vspace{0.4cm}
\textbf{Duality theory} provides:
\begin{enumerate}
    \item \textbf{Lower bounds} on the optimal value
    \item \textbf{Optimality conditions} (complementary slackness)
    \item \textbf{Alternative formulation} (often computationally useful)
    \item \textbf{Economic interpretation} (prices, equilibrium)
\end{enumerate}

\vspace{0.4cm}
\textbf{Key question:} What is the ``dual'' problem to (KP)?
\end{frame}

%==============================================================================
\begin{frame}{Deriving the Dual Problem}
\textbf{Key observation:} $\gamma \in \Pi(\mu, \nu)$ iff marginal constraints hold.

Encode constraints using Lagrange multipliers $\phi, \psi$:
\[
\sup_{\phi, \psi} \left[ \int_X \phi \, d\mu + \int_Y \psi \, d\nu - \int_{X \times Y} (\phi(x) + \psi(y)) \, d\gamma \right]
= \begin{cases} 0 & \gamma \in \Pi(\mu,\nu) \\ +\infty & \text{otherwise} \end{cases}
\]

\vspace{0.3cm}
\textbf{Rewrite (KP):}
\[
\min_\gamma \left[ \int c \, d\gamma + \sup_{\phi,\psi} \left( \int \phi \, d\mu + \int \psi \, d\nu - \int (\phi + \psi) \, d\gamma \right) \right]
\]

\vspace{0.3cm}
\textbf{Interchange min and sup} (minimax theorem):
\[
= \sup_{\phi,\psi} \left[ \int \phi \, d\mu + \int \psi \, d\nu + \underbrace{\inf_\gamma \int (c - \phi - \psi) \, d\gamma}_{=0 \text{ if } \phi+\psi \leq c, \; -\infty \text{ else}} \right]
\]
\end{frame}

%==============================================================================
\begin{frame}{The Dual Problem (DP)}
The inner $\inf_\gamma$ equals:
\[
\inf_{\gamma \geq 0} \int (c(x,y) - \phi(x) - \psi(y)) \, d\gamma =
\begin{cases}
0 & \text{if } \phi(x) + \psi(y) \leq c(x,y) \; \forall x,y \\
-\infty & \text{otherwise}
\end{cases}
\]

\vspace{0.3cm}
\[
\boxed{\textbf{(DP)} \quad \sup \left\{ \int_X \underbrace{\phi \, d\mu}_{\text{pay at source}} + \int_Y \underbrace{\psi \, d\nu}_{\text{receive at target}} \;:\; \underbrace{\phi(x) + \psi(y) \leq c(x,y)}_{\text{no arbitrage}} \right\}}
\]

\vspace{0.4cm}
\textbf{Economic interpretation:}
\begin{itemize}
    \item $\phi(x)$: price paid to pick up goods at $x$
    \item $\psi(y)$: price received for delivering goods at $y$
    \item Constraint: profit $\phi(x) + \psi(y) \leq$ transport cost $c(x,y)$
\end{itemize}
\end{frame}

%==============================================================================
\begin{frame}{Kantorovich Duality Theorem}
\begin{block}{Theorem (Kantorovich Duality)}
Under mild conditions:
\[
\boxed{\min_{\gamma \in \Pi(\mu,\nu)} \int c \, d\gamma = \sup_{\phi + \psi \leq c} \left( \int \phi \, d\mu + \int \psi \, d\nu \right)}
\]
\end{block}

\vspace{0.3cm}
\textbf{Key points:}
\begin{itemize}
    \item \textbf{Weak duality} ($\geq$): Easy to prove
    \item \textbf{Strong duality} ($=$): Requires proof (minimax or convex analysis)
    \item The supremum is often attained (under regularity of $c$)
\end{itemize}

\vspace{0.3cm}
\textbf{Optimality conditions:} $(\gamma^*, \phi^*, \psi^*)$ optimal iff:
\[
\phi^*(x) + \psi^*(y) = c(x,y) \quad \gamma^*\text{-almost everywhere}
\]
\end{frame}

\section{Summary}
%==============================================================================

\begin{frame}{Summary}
\textbf{Monge Problem (MP):}
\begin{itemize}
    \item Transport via maps $T: X \to Y$ with $T_\# \mu = \nu$
    \item Nonlinear constraint, may have no solution (mass can't split)
\end{itemize}

\vspace{0.2cm}
\textbf{Kantorovich Problem (KP):}
\begin{itemize}
    \item Transport via plans $\gamma \in \Pi(\mu, \nu)$
    \item Linear programming, always has solution (relaxation of MP)
\end{itemize}

\vspace{0.2cm}
\textbf{Existence Theory:}
\begin{itemize}
    \item Weierstrass + weak compactness + l.s.c. $\Rightarrow$ existence
    \item Works for Polish spaces with l.s.c. costs
\end{itemize}

\vspace{0.2cm}
\textbf{Duality:}
\[
\min_{\gamma \in \Pi(\mu,\nu)} \int c \, d\gamma = \sup_{\phi + \psi \leq c} \left( \int \phi \, d\mu + \int \psi \, d\nu \right)
\]

\vspace{0.2cm}
\textbf{Key Memo Boxes:} Weierstrass, Banach-Alaoglu, Riesz, Prokhorov, l.s.c. approximation, Lusin
\end{frame}
