\section{The Monge Problem}

%==============================================================================
% Slide 3: Monge Problem (1781)

\begin{frame}{The Monge Problem (1781)}
\begin{columns}
\begin{column}{0.55\textwidth}
\textbf{Historical Context:}
\begin{itemize}
    \item Gaspard Monge, France, before the Revolution
    \item Problem: Move sand from excavations (\textit{d\'eblais}) to constructions (\textit{remblais})
\end{itemize}

\vspace{0.3cm}
\textbf{Mathematical Formulation:}

Given densities $f, g \geq 0$ on $\R^d$ with $\int f = \int g = 1$, find $T: \R^d \to \R^d$ minimizing:
\[
M(T) := \int_{\R^d} |T(x) - x| f(x) \, dx
\]
subject to the \textbf{push-forward constraint}.
\end{column}

%==============================================================================
% Slide 4: Push-forward Measure
%==============================================================================
\begin{frame}{Push-forward Measure}
\begin{definition}[Image Measure]
For a measure $\mu$ on $X$ and a map $T: X \to Y$, the \textbf{push-forward} $\pushf{T}\mu$ is defined by:
\[
(\pushf{T}\mu)(A) = \mu(T^{-1}(A)) \quad \text{for every measurable set } A \subset Y
\]
\end{definition}

\vspace{0.3cm}
Equivalently, for every measurable function $\phi$:
\[
\int_Y \phi \, d(\pushf{T}\mu) = \int_X (\phi \circ T) \, d\mu
\]

%==============================================================================
% Slide 5: Monge Problem - General Formulation
%==============================================================================
\begin{frame}{General Monge Problem}
\textbf{General Cost Function:}

For a cost $c: X \times Y \to \R$, the \textbf{Monge Problem} (MP) is:
\[
\boxed{
\text{(MP)} \quad \min \left\{ M(T) := \int_X c(x, T(x)) \, d\mu(x) \;:\; \pushf{T}\mu = \nu \right\}
}
\]

\vspace{0.5cm}
\textbf{Jacobian Equation:}

If $\mu, \nu$ have densities $f, g$ and $T$ is smooth and injective:
\[
g(T(x)) \det(DT(x)) = f(x)
\]

\vspace{0.3cm}
\begin{alertblock}{Key Difficulty}
This equation is \textbf{highly nonlinear} in $T$, making existence proofs challenging.
\end{alertblock}
\end{frame}

%==============================================================================
% Slide 6: Difficulties with Monge Problem
%==============================================================================
\begin{frame}{Difficulties with the Monge Problem}
\textbf{Why is the Monge Problem Hard?}

\begin{enumerate}
    \item \textbf{Nonlinearity:} The constraint $\pushf{T}\mu = \nu$ is nonlinear
    \item \textbf{Non-existence:} Optimal map may not exist!
    \item \textbf{Weak convergence issues:} If $T_n \rightharpoonup T$, we cannot guarantee $\pushf{T_n}\mu \to \pushf{T}\mu$
\end{enumerate}

\vspace{0.5cm}
\begin{example}[Mass Splitting Required]
\begin{center}
\begin{tikzpicture}[scale=0.8]

    \fill[blue!70] (0,0) circle (5pt);

A transport \textbf{map} cannot split mass --- no solution exists!
\end{example}
\end{frame}

\section{Kantorovich Relaxation}
%==============================================================================

%==============================================================================
% Slide 7: Kantorovich's Idea (1942)
%==============================================================================
\begin{frame}{Kantorovich's Relaxation (1942)}
\textbf{Key Insight:} Replace transport \textit{maps} with transport \textit{plans}.

\vspace{0.3cm}
\begin{definition}[Transport Plan]
A \textbf{transport plan} is a measure $\gamma$ on $X \times Y$ with marginals:
\[
(\pi_x)_{\#}\gamma = \mu \quad \text{and} \quad (\pi_y)_{\#}\gamma = \nu
\]
The set of all such plans is denoted $\Pi(\mu, \nu)$.
\end{definition}

\vspace{0.3cm}
\textbf{Kantorovich Problem (KP):}
\[
\boxed{
\text{(KP)} \quad \min_{\gamma \in \Pi(\mu,\nu)} \int_{X \times Y} c(x,y) \, d\gamma(x,y)
}
\]

\begin{block}{Advantages}
\begin{itemize}
    \item Linear objective and constraints $\Rightarrow$ \textbf{Linear Programming}
    \item Existence of minimizer is guaranteed (weak-* compactness)
    \item Allows \textbf{mass splitting}
\end{itemize}
\end{block}
\end{frame}

\section{Brenier's Theorem}
%==============================================================================

%==============================================================================
% Slide 9: Brenier's Theorem
%==============================================================================
\begin{frame}{Brenier's Theorem (1991)}
\begin{theorem}[Brenier]
For the \textbf{quadratic cost} $c(x,y) = |x-y|^2$, if $\mu \ll \mathcal{L}^d$ (absolutely continuous):
\begin{enumerate}
    \item There exists a \textbf{unique} optimal transport map $T$
    \item $T = \nabla u$ for some \textbf{convex function} $u$
    \item The optimal plan is $\gamma = (\text{Id} \times \nabla u)_{\#}\mu$
\end{enumerate}
\end{theorem}

%==============================================================================
% Slide 10: Monge-Amp\`ere Equation
%==============================================================================
\begin{frame}{Connection to Monge-Amp\`ere Equation}
From the Jacobian equation with $T = \nabla u$:
\[
g(\nabla u(x)) \det(D^2 u(x)) = f(x)
\]

\vspace{0.3cm}
This is the \textbf{Monge-Amp\`ere equation}:
\[
\boxed{\det(D^2 u) = \frac{f(x)}{g(\nabla u(x))}}
\]

\vspace{0.3cm}
\begin{itemize}
    \item Degenerate elliptic PDE in the class of \textbf{convex functions}
    \item Rich regularity theory (Caffarelli, 1990s)
    \item Connection to differential geometry
\end{itemize}

\vspace{0.3cm}
\begin{block}{Polar Factorization (Brenier)}
Any vector field $v: \R^d \to \R^d$ can be written as:
\[
v = \nabla u \circ s
\]
where $u$ is convex and $s$ is measure-preserving.
\end{block}
\end{frame}

\section{Existence and Regularity}

%==============================================================================
% Slide 11: Sudakov's Approach

\begin{frame}{Sudakov's Approach (1979)}
\textbf{Strategy for the original Monge cost $c(x,y) = |x-y|$:}

\begin{enumerate}
    \item Decompose $\R^d$ into \textbf{transport rays} (line segments)
    \item These rays are preserved by the optimal plan $\gamma$
    \item Rays are level sets of a \textbf{Lipschitz function} (Kantorovich potential)
    \item Reduce to a family of \textbf{1D problems}
\end{enumerate}

%==============================================================================
% Slide 12: Regularity Theory
%==============================================================================
\begin{frame}{Regularity of Optimal Maps}
\textbf{Question:} When is the optimal map $T$ smooth?

\vspace{0.3cm}
\begin{theorem}[Caffarelli, 1990s --- Quadratic Cost]
If $\mu, \nu$ have smooth, strictly positive densities on convex domains, then $T = \nabla u$ is smooth.
\end{theorem}

\vspace{0.3cm}
\textbf{General Costs --- Ma-Trudinger-Wang Condition (2005):}

For $c \in C^4$, regularity holds if:
\[
\sum_{i,j,k,l,p,q,r,s} (c_{ij,rs} - c_{ij,p}c^{p,q}c_{q,rs})\xi_i\xi_j\eta_r\eta_s \geq 0
\]
for $\xi \perp \eta$ (in a suitable sense).

\vspace{0.3cm}
\begin{itemize}
    \item On Riemannian manifolds: related to \textbf{sectional curvature}
    \item Condition is \textbf{sharp} (Loeper, 2009)
\end{itemize}
\end{frame}

\section{Wasserstein Distance}
%==============================================================================

%==============================================================================
% Slide 13: Wasserstein Distance - Definition
%==============================================================================
\begin{frame}{Wasserstein Distance}
\begin{definition}[$p$-Wasserstein Distance]
For probability measures $\mu, \nu$ on $\R^d$ and $p \geq 1$:
\[
\boxed{
W_p(\mu, \nu) := \left( \inf_{\gamma \in \Pi(\mu,\nu)} \int_{X \times Y} |x-y|^p \, d\gamma(x,y) \right)^{1/p}
}
\]
\end{definition}

\vspace{0.3cm}
\textbf{Properties:}
\begin{itemize}
    \item $W_p$ is a \textbf{metric} on $\mathcal{P}_p(\R^d) = \{\mu : \int |x|^p d\mu < \infty\}$
    \item Metrizes \textbf{weak convergence} + convergence of $p$-th moments
    \item For $p=2$: $W_2^2(\mu,\nu) = \int |x - \nabla u(x)|^2 d\mu$ (Brenier)
\end{itemize}

\vspace{0.3cm}

\end{frame}

%==============================================================================
% Slide 14: Wasserstein Space Structure
%==============================================================================
\begin{frame}{Structure of Wasserstein Space}
\textbf{Geodesics in $(\mathcal{P}_2(\R^d), W_2)$:}

The geodesic from $\mu_0$ to $\mu_1$ is:
\[
\mu_t = ((1-t)\text{Id} + t\nabla u)_{\#}\mu_0, \quad t \in [0,1]
\]
where $\nabla u$ is the optimal map from $\mu_0$ to $\mu_1$.

\vspace{0.3cm}

\vspace{0.2cm}
\begin{block}{McCann's Displacement Convexity (1997)}
Many functionals $\mathcal{F}(\mu)$ are \textbf{convex along geodesics}:
\[
\mathcal{F}(\mu_t) \leq (1-t)\mathcal{F}(\mu_0) + t\mathcal{F}(\mu_1)
\]
\end{block}
\end{frame}

\section{Gradient Flows}
%==============================================================================

%==============================================================================
% Slide 15: Gradient Flows in Wasserstein Space
%==============================================================================
\begin{frame}{Gradient Flows in Wasserstein Space}
\textbf{Key Insight:} Many PDEs are \textbf{gradient flows} in $(\mathcal{P}_2, W_2)$.

\vspace{0.3cm}
\begin{example}[Heat Equation]
The heat equation $\partial_t \rho = \Delta \rho$ is the gradient flow of:
\[
\mathcal{F}(\rho) = \int \rho \log \rho \, dx \quad \text{(Entropy)}
\]
\end{example}

\vspace{0.3cm}
\textbf{JKO Scheme (Jordan-Kinderlehrer-Otto, 1998):}

Discrete-time approximation with time step $\tau$:
\[
\rho^{n+1} = \arg\min_{\rho} \left\{ \frac{1}{2\tau} W_2^2(\rho, \rho^n) + \mathcal{F}(\rho) \right\}
\]

\vspace{0.3cm}
\begin{itemize}
    \item Porous medium equation: $\mathcal{F}(\rho) = \frac{1}{m-1}\int \rho^m$
    \item Fokker-Planck equation: $\mathcal{F}(\rho) = \int \rho \log \rho + \int V \rho$
\end{itemize}
\end{frame}

\section{Geometry and Curvature}
%==============================================================================

%==============================================================================
% Slide 16: Entropy and Ricci Curvature
%==============================================================================
\begin{frame}{Entropy and Ricci Curvature}
\textbf{Remarkable Connection:}

On a Riemannian manifold $(M,g)$:
\[
\text{Entropy } E(\rho) = \int \rho \log \rho \text{ is geodesically convex in } W_2
\]
\[
\Updownarrow
\]
\[
\text{Ricci curvature } \geq 0
\]

\vspace{0.3cm}
\begin{block}{Synthetic Ricci Curvature (Lott-Villani, Sturm, 2006-2009)}
For metric measure spaces $(X, d, m)$, define:
\[
\text{Ric} \geq K \quad \Leftrightarrow \quad \text{Entropy is } K\text{-convex along } W_2\text{-geodesics}
\]
This extends Ricci curvature bounds to \textbf{non-smooth spaces}!
\end{block}

\vspace{0.2cm}
\textit{Major developments by Ambrosio, Gigli, Savar\'e, and others.}
\end{frame}

\begin{frame}{Applications: Economics and Traffic}
\textbf{Spatial Economics:}
\begin{itemize}
    \item Distribution of resources $\mu$ (supply)
    \item Distribution of demand $\nu$
    \item Optimal transport = efficient allocation
\end{itemize}

\vspace{0.3cm}
\textbf{Traffic Congestion Models:}

Modified cost depending on \textbf{traffic density}:
\[
\min \int_0^1 \int_\Omega g(\rho(t,x)) |\mathbf{v}(t,x)| \, dx \, dt
\]
where $g$ is increasing (congestion effect).

\vspace{0.3cm}
\textbf{Branched Transport:}

For building efficient networks, use \textbf{subadditive} costs:
\[
\text{cost}(m) = m^\alpha, \quad 0 < \alpha < 1
\]
Favors merging of transport paths $\Rightarrow$ tree-like structures.
\end{frame}

\section{Computational Methods}
%==============================================================================

%==============================================================================
% Slide 19: Computational Aspects
%==============================================================================
\begin{frame}{Computational Methods}
\textbf{1. Linear Programming:}
\begin{itemize}
    \item Discrete case: Hungarian algorithm, auction algorithm
    \item Complexity: $O(n^3)$ for $n$ points
\end{itemize}

\vspace{0.3cm}
\textbf{2. Entropic Regularization (Cuturi, 2013):}

Add entropy penalty:
\[
\min_{\gamma \in \Pi(\mu,\nu)} \int c \, d\gamma + \varepsilon \int \gamma \log \gamma
\]

\begin{itemize}
    \item \textbf{Sinkhorn algorithm}: Iterative scaling, very fast
    \item Complexity: nearly linear in $n$
    \item Differentiable $\Rightarrow$ machine learning applications
\end{itemize}

\vspace{0.3cm}
\textbf{3. Semi-discrete Methods:}
\begin{itemize}
    \item One measure continuous, one discrete
    \item Solve via Laguerre tessellation / power diagrams
\end{itemize}
\end{frame}
