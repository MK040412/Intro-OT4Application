\end{center}

\vspace{0.5cm}

\begin{tikzpicture}
    \node[draw,rounded corners,fill=yellow!20,text width=6cm,align=center] {
        \textbf{Conservation of Mass:}\\[3pt]
        $\mu(T^{-1}(A)) = \nu(A)$\\[3pt]
        {\small Mass leaving $T^{-1}(A)$ = Mass arriving at $A$}
    };
\end{tikzpicture}
\end{column}
\begin{column}{0.5\textwidth}
\textbf{Formal Definition:}
\[
(T_\#\mu)(A) := \mu(T^{-1}(A))
\]

\vspace{0.3cm}
\textbf{Equivalent form:}
\[
\int_Y \phi(y) \, d(T_\#\mu) = \int_X \phi(T(x)) \, d\mu
\]

\vspace{0.3cm}
\begin{alertblock}{Key Point}
$T_\# \mu = \nu$ means:\\
``$T$ moves $\mu$ exactly onto $\nu$''
\end{alertblock}
\end{column}
\end{columns}
\end{frame}

\begin{frame}{General Monge Problem}
\begin{center}
\begin{tikzpicture}
    % Main equation box
    \node[draw,very thick,rounded corners,fill=gray!10,inner sep=12pt] (main) {
        $\displaystyle \text{(MP)} \quad \min_{T} \left\{ \int_X c(x, T(x)) \, d\mu(x) \;\Big|\; T_\#\mu = \nu \right\}$
    };
\end{tikzpicture}

\vspace{0.6cm}

\begin{tikzpicture}[scale=0.9]
    % Annotations
    \node[draw,rounded corners,fill=blue!15,text width=2.8cm,align=center] (c1) at (-5,0) {
        \textbf{Cost $c(x,y)$}\\[2pt]
        {\small Cost to move from $x$ to $y$}
    };

    \node[draw,rounded corners,fill=green!15,text width=3cm,align=center] (c2) at (-1.2,0) {
        \textbf{Integral}\\[2pt]
        {\small Sum over all source points}
    };

    \node[draw,rounded corners,fill=red!15,text width=3cm,align=center] (c3) at (3,0) {
        \textbf{Constraint}\\[2pt]
        {\small Mass conservation}
    };

    \node[draw,rounded corners,fill=purple!15,text width=2.5cm,align=center] (c4) at (6.5,0) {
        \textbf{min over $T$}\\[2pt]
        {\small Find best map}
    };
\end{tikzpicture}
\end{center}

\vspace{0.5cm}

\textbf{Common cost functions:}
\begin{itemize}
    \item $c(x,y) = |x-y|$ \quad (Monge's original --- distance)
    \item $c(x,y) = |x-y|^2$ \quad (Quadratic --- kinetic energy)
    \item $c(x,y) = -\log|x-y|$ \quad (Logarithmic)
\end{itemize}
\end{frame}

    \fill[red!70] (4,-0.8) circle (5pt);

    % Impossible arrows
    \draw[-{Stealth},thick,gray,dashed] (0.3,0.1) -- (3.7,0.7);

    % X mark
    \node[red,font=\Huge] at (2,0) {\texttimes};
\end{tikzpicture}

\vspace{0.3cm}

\begin{alertblock}{No Solution Exists!}
A \textit{function} $T$ must send $0$ to a single point, but we need to split the mass to two destinations.
\end{alertblock}
\end{column}
\begin{column}{0.45\textwidth}
\textbf{Mathematical Issues:}

\begin{enumerate}
    \item \textbf{Non-convex:}\\
    {\small Set of valid maps is not convex}

    \vspace{0.3cm}
    \item \textbf{Non-existence:}\\
    {\small Minimizer may not exist}

    \vspace{0.3cm}
    \item \textbf{Nonlinear constraint:}\\
    {\small $T_\#\mu = \nu$ is hard to handle}
\end{enumerate}

\vspace{0.5cm}
\begin{block}{Historical Note}
Problem remained unsolved for $\sim$160 years!
\end{block}
\end{column}
\end{columns}
\end{frame}

    % Marginal arrows
    \draw[-{Stealth},blue,thick] (2.5,5.2) -- (2.5,5.8) node[above] {$\pi_x \to \mu$};
    \draw[-{Stealth},green!60!black,thick] (5.2,2.5) -- (5.8,2.5) node[right] {$\pi_y \to \nu$};
\end{tikzpicture}
\end{center}
\end{column}
\begin{column}{0.5\textwidth}
\textbf{Interpretation:}
\begin{itemize}
    \item $\gamma(x,y)$ = amount of mass transported from $x$ to $y$
    \item Row sums = source $\mu$
    \item Column sums = target $\nu$
\end{itemize}

\vspace{0.3cm}
\textbf{Key property:}
\[
\int_Y d\gamma(x,y) = d\mu(x)
\]
\[
\int_X d\gamma(x,y) = d\nu(y)
\]
\end{column}
\end{columns}
\end{frame}

\begin{frame}{The Kantorovich Problem}
\begin{center}
\begin{tikzpicture}
    \node[draw,very thick,rounded corners,fill=gray!10,inner sep=15pt,scale=1.1] {
        $\displaystyle \text{(KP)} \quad \min_{\gamma \in \Pi(\mu,\nu)} \int_{X \times Y} c(x,y) \, d\gamma(x,y)$
    };
\end{tikzpicture}

\vspace{0.8cm}

\begin{tikzpicture}[scale=0.85]
    % Left box
    \node[draw,rounded corners,fill=blue!15,text width=4cm,align=center] at (-4,0) {
        \textbf{Linear in $\gamma$}\\[3pt]
        {\small Objective is linear!}\\[3pt]
        $\Rightarrow$ \textbf{Convex optimization}
    };

    % Middle box
    \node[draw,rounded corners,fill=green!15,text width=4cm,align=center] at (0,0) {
        \textbf{Linear constraints}\\[3pt]
        {\small Marginal conditions are linear}\\[3pt]
        $\Rightarrow$ \textbf{Linear Programming}
    };

    % Right box
    \node[draw,rounded corners,fill=red!15,text width=4cm,align=center] at (4,0) {
        \textbf{Existence guaranteed}\\[3pt]
        {\small Weak-* compactness}\\[3pt]
        $\Rightarrow$ \textbf{Solution exists!}
    };
\end{tikzpicture}
\end{center}

\vspace{0.5cm}

\begin{block}{Monge $\subset$ Kantorovich}
If $T$ is a Monge map, then $\gamma = (\text{Id} \times T)_\# \mu$ is a Kantorovich plan.\\
So: \quad $\min(\text{KP}) \leq \min(\text{MP})$
\end{block}
\end{frame}

\section{Brenier's Theorem}
%==============================================================================

%==============================================================================
% Slide 11: Brenier's Theorem - Statement
%==============================================================================
\begin{frame}{Brenier's Theorem (1991)}
\begin{center}
\begin{tikzpicture}
    \node[draw,very thick,rounded corners,fill=blue!10,inner sep=12pt] {
        \textbf{Quadratic Cost:} \quad $c(x,y) = |x-y|^2$
    };
\end{tikzpicture}
\end{center}

\vspace{0.3cm}

\begin{theorem}[Brenier]
If $\mu \ll \mathcal{L}^d$ (absolutely continuous), then:
\begin{enumerate}
    \item There exists a \textbf{unique} optimal transport map $T$
    \item $T = \nabla u$ for some \textbf{convex function} $u: \R^d \to \R$
\end{enumerate}
\end{theorem}

\vspace{0.3cm}

\begin{center}
\begin{tikzpicture}[scale=0.8]
    % Convex function
    \begin{scope}
        \draw[thick,blue!70,fill=blue!10] plot[smooth,domain=-2:2] (\x,{0.3*\x*\x}) -- (2,0) -- (-2,0) -- cycle;
        \draw[thick,blue!70] plot[smooth,domain=-2:2] (\x,{0.3*\x*\x});
        \node[below] at (0,-0.3) {Convex $u(x)$};
    \end{scope}

    % Arrow
    \draw[-{Stealth},very thick] (2.5,0.6) -- node[above] {$\nabla$} (4,0.6);

    % Gradient = Transport
    \begin{scope}[shift={(6,0.6)}]
        \foreach \x in {-1.5,-0.75,0,0.75,1.5} {
            \draw[-{Stealth},red!70,thick] (\x,0) -- ({\x+0.5*\x},0);
        }
        \node[below] at (0,-0.8) {$T(x) = \nabla u(x)$};
    \end{scope}
\end{tikzpicture}
\end{center}

\begin{alertblock}{Remarkable}
The optimal transport is the \textbf{gradient of a convex function}!
\end{alertblock}
\end{frame}

    \node[orange,font=\bfseries] at (0,-2.5) {$T = \nabla u$ pushes $\mu$ to $\nu$ optimally};
\end{tikzpicture}
\end{center}

\vspace{0.3cm}

\textbf{Why convex?}
\begin{itemize}
    \item Gradient of convex function is \textbf{monotone}: $\langle \nabla u(x) - \nabla u(y), x-y \rangle \geq 0$
    \item This prevents ``crossing'' of transport paths
    \item Crossing would waste energy!
\end{itemize}
\end{frame}

\begin{tikzpicture}[scale=0.8]

        ``Earth Mover's Distance''
    };

    \node[draw,rounded corners,fill=orange!15,text width=3.5cm,align=center] at (0,0) {
        \textbf{$|x-y|^p$}\\[3pt]
        Cost of moving\\
        unit mass from $x$ to $y$
    };

    \node[draw,rounded corners,fill=green!15,text width=3.5cm,align=center] at (5,0) {
        \textbf{$(\cdot)^{1/p}$}\\[3pt]
        Normalization to\\
        get a metric
    };
\end{tikzpicture}

\vspace{0.5cm}

\begin{columns}
\begin{column}{0.5\textwidth}
\begin{tikzpicture}[scale=0.6]
    \fill[blue!40] (0,0) ellipse (0.8 and 0.5);
    \node[below,blue!70] at (0,-0.8) {$\mu$};

    \fill[red!40] (4,0) ellipse (0.6 and 0.7);
    \node[below,red!70] at (4,-0.9) {$\nu$};

    \draw[<->,thick,dashed,purple] (0.9,0) -- node[above] {$W_p$} (3.3,0);
\end{tikzpicture}
\end{column}
\begin{column}{0.5\textwidth}
\textbf{Most common:} $p=2$
\[
W_2^2(\mu,\nu) = \min_\gamma \int |x-y|^2 d\gamma
\]
(Related to kinetic energy)
\end{column}
\end{columns}
\end{frame}

%==============================================================================
% Slide 15: Wasserstein is a Metric
%==============================================================================
\begin{frame}{$W_p$ is a True Distance}
\begin{theorem}
$W_p$ is a \textbf{metric} on $\mathcal{P}_p(\R^d) = \{\mu : \int |x|^p d\mu < \infty\}$
\end{theorem}

\vspace{0.3cm}

\begin{center}
\begin{tikzpicture}[scale=0.85]
    % Three properties
    \node[draw,rounded corners,fill=blue!15,text width=4cm,align=center,minimum height=2cm] at (-4.5,0) {
        \textbf{1. Non-negative}\\[5pt]
        $W_p(\mu,\nu) \geq 0$\\
        $W_p(\mu,\nu) = 0 \Leftrightarrow \mu=\nu$
    };

    \node[draw,rounded corners,fill=green!15,text width=4cm,align=center,minimum height=2cm] at (0,0) {
        \textbf{2. Symmetric}\\[5pt]
        $W_p(\mu,\nu) = W_p(\nu,\mu)$
    };

    \node[draw,rounded corners,fill=red!15,text width=4cm,align=center,minimum height=2cm] at (4.5,0) {
        \textbf{3. Triangle inequality}\\[5pt]
        $W_p(\mu,\rho) \leq$\\
        $W_p(\mu,\nu) + W_p(\nu,\rho)$
    };
\end{tikzpicture}
\end{center}

\vspace{0.5cm}

\begin{block}{Key Property}
$W_p$ metrizes weak convergence + convergence of moments:
\[
W_p(\mu_n, \mu) \to 0 \quad \Leftrightarrow \quad \mu_n \rightharpoonup \mu \text{ and } \int|x|^p d\mu_n \to \int|x|^p d\mu
\]
\end{block}
\end{frame}

\begin{frame}{JKO Scheme: Discrete Gradient Flow}
\textbf{Jordan-Kinderlehrer-Otto (1998):}

\begin{center}
\begin{tikzpicture}
    \node[draw,very thick,rounded corners,fill=blue!10,inner sep=12pt] {
        $\displaystyle \rho^{n+1} = \arg\min_\rho \left\{ \frac{1}{2\tau} W_2^2(\rho, \rho^n) + \mathcal{F}(\rho) \right\}$
    };
\end{tikzpicture}
\end{center}

\vspace{0.5cm}

\begin{tikzpicture}[scale=0.85]
    \node[draw,rounded corners,fill=green!15,text width=4.5cm,align=center] at (-4,0) {
        \textbf{$\frac{1}{2\tau}W_2^2(\rho,\rho^n)$}\\[5pt]
        {\small ``Don't move too far from $\rho^n$''}\\
        (Proximal term)
    };

    \node[draw,rounded corners,fill=red!15,text width=4.5cm,align=center] at (2,0) {
        \textbf{$\mathcal{F}(\rho)$}\\[5pt]
        {\small ``Minimize the energy''}\\
        (Driving functional)
    };

    \node[draw,rounded corners,fill=purple!15,text width=3cm,align=center] at (6.5,0) {
        \textbf{Balance}\\[5pt]
        {\small Small $\tau$: stay close\\
        Large $\tau$: minimize $\mathcal{F}$}
    };
\end{tikzpicture}

\vspace{0.5cm}

\begin{block}{Convergence}
As $\tau \to 0$, the discrete scheme converges to the continuous gradient flow:
\[
\rho^n \;\longrightarrow\; \rho(n\tau) \quad \text{solves } \partial_t\rho = -\text{grad}_{W_2}\mathcal{F}
\]
\end{block}
\end{frame}
